  (function () {
    'use strict';

    var reduceQuery = window.matchMedia('(prefers-reduced-motion: reduce)');
    var prefersReduced = reduceQuery.matches;

    /* If reduced motion, mark html.no-motion so CSS reveals everything. */
    if (prefersReduced) {
      document.documentElement.classList.add('no-motion');
    }

    /* ------------------------------------------------------------------ */
    /* Navigation: theme switcher                                          */
    /* ------------------------------------------------------------------ */

    function initThemeSwitcher() {
      var Theme = window.PortfolioTheme;
      if (!Theme) { return; }

      var buttons = document.querySelectorAll('.nav__theme-btn');

      function syncButtons(active) {
        Array.prototype.forEach.call(buttons, function (btn) {
          var name = btn.getAttribute('data-theme-value');
          btn.setAttribute('aria-pressed', name === active ? 'true' : 'false');
        });
      }

      Array.prototype.forEach.call(buttons, function (btn) {
        btn.addEventListener('click', function () {
          var name = btn.getAttribute('data-theme-value');
          if (Theme.isSupported(name)) {
            Theme.set(name);
          }
        });
      });

      syncButtons(Theme.current() || 'dark');

      document.addEventListener('themechange', function (event) {
        syncButtons(event.detail && event.detail.theme
          ? event.detail.theme
          : 'dark');
      });
    }

    /* ------------------------------------------------------------------ */
    /* Navigation: mobile menu                                             */
    /* ------------------------------------------------------------------ */

    function initMobileMenu() {
      var nav = document.getElementById('nav');
      var toggle = document.getElementById('navToggle');
      var links = document.getElementById('navLinks');
      if (!nav || !toggle || !links) { return; }

      function close() {
        nav.classList.remove('is-open');
        document.body.classList.remove('menu-open');
        toggle.setAttribute('aria-expanded', 'false');
        toggle.setAttribute('aria-label', 'Open menu');
      }

      function open() {
        nav.classList.add('is-open');
        document.body.classList.add('menu-open');
        toggle.setAttribute('aria-expanded', 'true');
        toggle.setAttribute('aria-label', 'Close menu');
      }

      toggle.addEventListener('click', function () {
        if (nav.classList.contains('is-open')) { close(); } else { open(); }
      });

      Array.prototype.forEach.call(links.querySelectorAll('a'), function (a) {
        a.addEventListener('click', close);
      });

      document.addEventListener('keydown', function (event) {
        if (event.key === 'Escape' && nav.classList.contains('is-open')) {
          close();
          toggle.focus();
        }
      });

      /* Close menu when viewport grows beyond mobile breakpoint. */
      window.addEventListener('resize', function () {
        if (window.innerWidth > 780) { close(); }
      });
    }

    /* ------------------------------------------------------------------ */
    /* Nav background on scroll                                            */
    /* ------------------------------------------------------------------ */

    function initNavScroll() {
      var nav = document.getElementById('nav');
      var progress = document.getElementById('progressBar');
      if (!nav) { return; }

      var ticking = false;

      function update() {
        ticking = false;
        var scrollY = window.scrollY || window.pageYOffset;
        var maxScroll = Math.max(1,
          document.documentElement.scrollHeight - window.innerHeight);

        if (scrollY > 12) {
          nav.classList.add('is-scrolled');
        } else {
          nav.classList.remove('is-scrolled');
        }

        if (progress) {
          var pct = Math.min(1, scrollY / maxScroll);
          progress.style.transform = 'scaleX(' + pct.toFixed(4) + ')';
        }
      }

      window.addEventListener('scroll', function () {
        if (!ticking) {
          ticking = true;
          window.requestAnimationFrame(update);
        }
      }, { passive: true });

      update();
    }

    /* ------------------------------------------------------------------ */
    /* GSAP animations                                                     */
    /* ------------------------------------------------------------------ */

    function initAnimations() {
      if (prefersReduced) { return; }

      var gsapLib = window.gsap;
      var ScrollTriggerLib = window.ScrollTrigger;

      if (!gsapLib || !ScrollTriggerLib) {
        /* Fallback: show all reveal content if GSAP failed to load. */
        document.documentElement.classList.add('no-motion');
        return;
      }

      gsapLib.registerPlugin(ScrollTriggerLib);

      /* ---- Reveal on scroll (single-shot per element) ---- */
      var revealEls = document.querySelectorAll('[data-reveal]');
      Array.prototype.forEach.call(revealEls, function (el) {
        gsapLib.to(el, {
          opacity: 1,
          y: 0,
          duration: 0.95,
          ease: 'power3.out',
          scrollTrigger: {
            trigger: el,
            start: 'top 88%',
            once: true
          }
        });
      });

      /* ---- Section heads get a slightly richer reveal ---- */
      var headEls = document.querySelectorAll('.section__head [data-reveal]');
      Array.prototype.forEach.call(headEls, function (el, i) {
        gsapLib.fromTo(el,
          { opacity: 0, y: 24 },
          {
            opacity: 1,
            y: 0,
            duration: 0.9,
            delay: i * 0.06,
            ease: 'power3.out',
            scrollTrigger: {
              trigger: el.closest('.section__head'),
              start: 'top 85%',
              once: true
            }
          }
        );
      });

      /* ---- Philosophy quote: character fade for extra weight ---- */
      var quote = document.querySelector('.philosophy__quote');
      if (quote) {
        gsapLib.fromTo(quote,
          { opacity: 0, y: 28 },
          {
            opacity: 1,
            y: 0,
            duration: 1.1,
            ease: 'power3.out',
            scrollTrigger: {
              trigger: '.philosophy',
              start: 'top 70%',
              once: true
            }
          }
        );
      }

      /* ---- Approach cards: staggered lift ---- */
      var approachCards = document.querySelectorAll('.approach__card');
      if (approachCards.length) {
        gsapLib.fromTo(approachCards,
          { opacity: 0, y: 30 },
          {
            opacity: 1,
            y: 0,
            duration: 0.8,
            stagger: 0.08,
            ease: 'power3.out',
            scrollTrigger: {
              trigger: '.approach',
              start: 'top 80%',
              once: true
            }
          }
        );
      }

      /* ---- Interest items: staggered fade ---- */
      var interestItems = document.querySelectorAll('.interest');
      if (interestItems.length) {
        gsapLib.fromTo(interestItems,
          { opacity: 0, y: 24 },
          {
            opacity: 1,
            y: 0,
            duration: 0.75,
            stagger: 0.06,
            ease: 'power2.out',
            scrollTrigger: {
              trigger: '.interests__grid',
              start: 'top 82%',
              once: true
            }
          }
        );
      }

      /* ---- Journey cards: staggered fade ---- */
      var journeyCards = document.querySelectorAll('.journey__card');
      if (journeyCards.length) {
        gsapLib.fromTo(journeyCards,
          { opacity: 0, y: 30 },
          {
            opacity: 1,
            y: 0,
            duration: 0.85,
            stagger: 0.1,
            ease: 'power3.out',
            scrollTrigger: {
              trigger: '.journey__grid',
              start: 'top 82%',
              once: true
            }
          }
        );
      }

      /* ---- Background orbs: subtle parallax (desktop only) ---- */
      var orbs = document.querySelectorAll('[data-orb]');
      var isDesktop = window.matchMedia('(min-width: 780px)').matches;
      var isCoarse = window.matchMedia('(pointer: coarse)').matches;

      if (orbs.length && isDesktop && !isCoarse && gsapLib.quickTo) {
        var depths = [18, 26, 12];
        var setters = [];

        for (var i = 0; i < orbs.length; i += 1) {
          setters.push({
            x: gsapLib.quickTo(orbs[i], 'x', { duration: 1.4, ease: 'power3.out' }),
            y: gsapLib.quickTo(orbs[i], 'y', { duration: 1.4, ease: 'power3.out' }),
            depth: depths[i] || 18
          });
        }

        var frame = null;
        var px = 0, py = 0;

        window.addEventListener('pointermove', function (event) {
          px = (event.clientX / window.innerWidth - 0.5) * 2;
          py = (event.clientY / window.innerHeight - 0.5) * 2;

          if (frame) { return; }
          frame = window.requestAnimationFrame(function () {
            frame = null;
            for (var j = 0; j < setters.length; j += 1) {
              setters[j].x(px * setters[j].depth);
              setters[j].y(py * setters[j].depth);
            }
          });
        }, { passive: true });
      }
    }

        }
      }, 2000);
    }

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', boot);
    } else {
      boot();
    }
  })();

/* ==========================================================================
   Aman Sohail — Portfolio
   assets/js/theme.js
   --------------------------------------------------------------------------
   Shared theme system. Loaded in <head> on every page so the persisted
   theme is applied to <html data-theme="..."> before first paint.

   Public API (window.PortfolioTheme):
     .KEY                 storage key used for persistence
     .ATTRIBUTE           attribute applied to <html>
     .list                ['dark', 'rich', 'cyberpunk']
     .isSupported(name)   boolean
     .info(name)          { label, color, description } | null
     .label(name)         string
     .get()               persisted theme or null
     .set(name)           persist + apply, returns applied theme
     .apply(name)         apply only (no persistence)
     .clear()             remove preference and reset
     .current()           theme currently applied to <html>
   ========================================================================== */

(function (window, document) {
  'use strict';

  var STORAGE_KEY = 'amanSohail:theme';
  var ATTRIBUTE = 'data-theme';

  var THEMES = {
    dark: {
      label: 'Dark',
      color: '#0a0a0c',
      description: 'Minimal, focused and professional.'
    },
    rich: {
      label: 'Rich',
      color: '#150b26',
      description: 'Vibrant, expressive and colourful.'
    },
    cyberpunk: {
      label: 'Cyberpunk',
      color: '#04060f',
      description: 'Neon, futuristic and immersive.'
    }
  };

  var ORDER = ['dark', 'rich', 'cyberpunk'];

  /* ---------------------------------------------------------------------- */

  function isSupported(value) {
    return typeof value === 'string' &&
           Object.prototype.hasOwnProperty.call(THEMES, value);
  }

  function readPreference() {
    try {
      var stored = window.localStorage.getItem(STORAGE_KEY);
      return isSupported(stored) ? stored : null;
    } catch (error) {
      return null; /* private mode / storage disabled */
    }
  }

  function writePreference(value) {
    try {
      window.localStorage.setItem(STORAGE_KEY, value);
      return true;
    } catch (error) {
      return false;
    }
  }

  function removePreference() {
    try {
      window.localStorage.removeItem(STORAGE_KEY);
    } catch (error) {
      /* ignore */
    }
  }

  function updateMetaColor(theme) {
    var meta = document.querySelector('meta[name="theme-color"]');
    if (meta && theme) {
      meta.setAttribute('content', THEMES[theme].color);
    }
  }

  function emit(theme) {
    var event;
    try {
      event = new CustomEvent('themechange', {
        detail: {
          theme: theme,
          label: theme ? THEMES[theme].label : null
        }
      });
    } catch (error) {
      /* CustomEvent constructor unavailable — degrade silently */
      return;
    }
    document.dispatchEvent(event);
  }

  /* ---------------------------------------------------------------------- */

  function apply(value) {
    var theme = isSupported(value) ? value : null;
    var root = document.documentElement;

    if (theme) {
      root.setAttribute(ATTRIBUTE, theme);
      root.classList.add('has-theme');
    } else {
      root.removeAttribute(ATTRIBUTE);
      root.classList.remove('has-theme');
    }

    updateMetaColor(theme);
    emit(theme);

    return theme;
  }

  function set(value) {
    var theme = apply(value);
    if (theme) {
      writePreference(theme);
    } else {
      removePreference();
    }
    return theme;
  }

  function current() {
    var applied = document.documentElement.getAttribute(ATTRIBUTE);
    return isSupported(applied) ? applied : null;
  }

  /* ---------------------------------------------------------------------- */

  /* Apply the stored preference immediately — before the first paint. */
  var initial = readPreference();
  if (initial) {
    apply(initial);
  }

  window.PortfolioTheme = {
    KEY: STORAGE_KEY,
    ATTRIBUTE: ATTRIBUTE,
    list: ORDER.slice(),
    isSupported: isSupported,
    info: function (name) {
      return isSupported(name) ? THEMES[name] : null;
    },
    label: function (name) {
      return isSupported(name) ? THEMES[name].label : '';
    },
    get: readPreference,
    set: set,
    apply: apply,
    clear: function () {
      removePreference();
      return apply(null);
    },
    current: current
  };
})(window, document);

/* ==========================================================================

   assets/js/welcome.js
   --------------------------------------------------------------------------

   confirmation dock, persistence hand-off and entry navigation.
   ========================================================================== */

(function (window, document) {
  'use strict';

  var Theme = window.PortfolioTheme;
  if (!Theme) {
    /* Theme system missing — do not trap the visitor behind the loader. */
    var stuck = document.getElementById('loader');
    if (stuck) { stuck.parentNode.removeChild(stuck); }
    return;
  }

  /* ---------------------------------------------------------------------- */
  /* Environment                                                            */
  /* ---------------------------------------------------------------------- */

  var reduceQuery = window.matchMedia('(prefers-reduced-motion: reduce)');
  var prefersReduced = reduceQuery.matches;

  var gsapLib = window.gsap;
  var motionOK = !prefersReduced &&
                 !!gsapLib &&
                 typeof gsapLib.timeline === 'function';

  var ENTRY_URL = 'home.html';

  var dom = {};
  var state = {
    selected: null,
    dockVisible: false,
    entered: false
  };

  /* ---------------------------------------------------------------------- */
  /* Helpers                                                                */
  /* ---------------------------------------------------------------------- */

  function $(selector, context) {
    return (context || document).querySelector(selector);
  }

  function $$(selector, context) {
    return Array.prototype.slice.call(
      (context || document).querySelectorAll(selector)
    );
  }

  function pad(value) {
    return (value < 10 ? '0' : '') + value;
  }

  function cardFor(name) {
    for (var i = 0; i < dom.cards.length; i += 1) {
      if (dom.cards[i].getAttribute('data-theme-value') === name) {
        return dom.cards[i];
      }
    }
    return null;
  }

  /* ---------------------------------------------------------------------- */
  /* DOM cache                                                              */
  /* ---------------------------------------------------------------------- */

  function cacheDom() {
    dom.loader = $('#loader');
    dom.loaderFill = $('.loader__fill');
    dom.loaderPct = $('.loader__pct span');
    dom.page = $('#page');
    dom.cardsWrap = $('#cards');
    dom.cards = $$('.card');
    dom.status = $('#status');
    dom.dock = $('#dock');
    dom.dockTitle = $('#dockTitle');
    dom.dockSub = $('#dockSub');
    dom.enterBtn = $('#enterBtn');
    dom.changeBtn = $('#changeBtn');
    dom.flash = $('#flash');
    dom.orbs = $$('.bg__orb');
  }

  /* ---------------------------------------------------------------------- */
  /* 1. Loader                                                              */
  /* ---------------------------------------------------------------------- */

  function runLoader() {
    return new Promise(function (resolve) {
      var loader = dom.loader;

      if (!loader) { resolve(); return; }

      var settled = false;

      function finish() {
        if (settled) { return; }
        settled = true;

        loader.classList.add('is-done');
        loader.setAttribute('aria-hidden', 'true');
        document.body.classList.remove('is-loading');

        /* Remove from the layout once the fade has completed. */
        window.setTimeout(function () {
          if (loader.parentNode) {
            loader.parentNode.removeChild(loader);
          }
        }, 700);

        resolve();
      }

      /* Failsafe: never hold the visitor hostage. */
      window.setTimeout(finish, 4500);

      if (!motionOK) {
        if (dom.loaderPct) { dom.loaderPct.textContent = '100'; }
        window.setTimeout(finish, 280);
        return;
      }

      var counter = { value: 0 };

      var timeline = gsapLib.timeline({ onComplete: finish });

      timeline
        .from('.loader__mark', {
          y: 14, opacity: 0, duration: 0.7, ease: 'power3.out'
        }, 0.05)
        .from('.loader__track', {
          opacity: 0, scaleX: 0.45, duration: 0.7, ease: 'power3.out'
        }, 0.12)
        .to(dom.loaderFill, {
          scaleX: 1, duration: 0.95, ease: 'power2.inOut'
        }, 0.18)
        .to(counter, {
          value: 100,
          duration: 0.95,
          ease: 'power2.inOut',
          onUpdate: function () {
            if (dom.loaderPct) {
              dom.loaderPct.textContent = pad(Math.round(counter.value));
            }
          }
        }, 0.18)
        .to('.loader__inner', {
          opacity: 0, y: -10, duration: 0.42, ease: 'power2.in'
        }, '+=0.12')
        .to(loader, {
          opacity: 0, duration: 0.5, ease: 'power2.inOut'
        }, '-=0.2');
    });
  }

  /* ---------------------------------------------------------------------- */
  /* 2. Entrance reveal                                                     */
  /* ---------------------------------------------------------------------- */

  function reveal() {
    if (!motionOK) {
      document.body.classList.remove('is-booting');
      return;
    }

    var timeline = gsapLib.timeline({
      defaults: { ease: 'power3.out' },
      onComplete: function () {
        document.body.classList.remove('is-booting');
        gsapLib.set('.topbar, .intro [data-anim], .card, .footnote', {
          clearProps: 'opacity,transform'
        });
      }

    timeline
      .from('.topbar', { y: -14, opacity: 0, duration: 0.8 }, 0)
      .from('.intro [data-anim]', {
        y: 26, opacity: 0, duration: 0.95, stagger: 0.09
      }, 0.14)
      .from('.card', {
        y: 40, opacity: 0, duration: 0.9, stagger: 0.09

      .from('.footnote', { opacity: 0, duration: 0.8 }, 0.78);
  }

  /* ---------------------------------------------------------------------- */
  /* 3. Theme selection                                                     */
  /* ---------------------------------------------------------------------- */

  function selectTheme(name, options) {
    options = options || {};

    var meta = Theme.info(name);
    if (!meta) { return; }

    if (state.selected === name && !options.force) {
      showDock(name, options);
      return;
    }

    state.selected = name;

    dom.cards.forEach(function (card) {
      var active = card.getAttribute('data-theme-value') === name;

    /* Applies the theme, persists it and dispatches 'themechange'. */
    Theme.set(name);

    updateStatus(name, options.returning);

    if (!options.silent) {
      bloom(name);

    showDock(name, { animate: !options.silent, returning: options.returning });

  function bloom(name) {
    if (!motionOK) { return; }

    var card = cardFor(name);
    if (!card) { return; }

    if (dom.flash) {
      var rect = card.getBoundingClientRect();
      var fx = ((rect.left + rect.width / 2) / window.innerWidth) * 100;
      var fy = ((rect.top + rect.height / 2) / window.innerHeight) * 100;

      dom.flash.style.setProperty('--fx', fx.toFixed(2) + '%');
      dom.flash.style.setProperty('--fy', fy.toFixed(2) + '%');

      gsapLib.fromTo(dom.flash, { opacity: 0 }, {
        opacity: 0.42,
        duration: 0.32,
        ease: 'power2.out',
        onComplete: function () {
          gsapLib.to(dom.flash, {
            opacity: 0, duration: 0.9, ease: 'power2.in'
          });
        }
      });
    }

    gsapLib.fromTo(card, { scale: 1 }, {
      scale: 1.025,
      duration: 0.32,
      ease: 'power2.out',
      yoyo: true,
      repeat: 1,
      transformOrigin: '50% 50%',
      onComplete: function () {
        gsapLib.set(card, { clearProps: 'transform' });
      }
    });
  }

  function updateStatus(name, returning) {
    if (!dom.status) { return; }

    if (!name) {
      dom.status.textContent = '';
      dom.status.classList.remove('is-visible');
      return;

    var label = Theme.label(name);

    dom.status.textContent = returning
      ? label + ' experience restored from your last visit'
      : 'Selected: ' + label;

    dom.status.classList.add('is-visible');
  }

  /* ---------------------------------------------------------------------- */
  /* 4. Confirmation dock                                                   */
  /* ---------------------------------------------------------------------- */

  function showDock(name, options) {
    options = options || {};
    if (!dom.dock) { return; }

    var meta = Theme.info(name);

    dom.dock.hidden = false;
    document.body.classList.add('dock-open');
    state.dockVisible = true;

    if (dom.dockTitle) {
      dom.dockTitle.textContent = meta.label + ' experience selected';
    }
    if (dom.dockSub) {
      dom.dockSub.textContent = options.returning
        ? 'Restored from your last visit · change it anytime'
        : 'Saved to this device · change it anytime';
    }

    if (options.animate && motionOK) {
      gsapLib.fromTo(dom.dock, { y: 34, opacity: 0 }, {
        y: 0,
        opacity: 1,
        duration: 0.85,

      });

  }

  function hideDock(onDone) {
    if (!dom.dock || dom.dock.hidden) {
      if (typeof onDone === 'function') { onDone(); }
      return;
    }

    state.dockVisible = false;
    document.body.classList.remove('dock-open');

    function complete() {
      dom.dock.hidden = true;
      if (typeof onDone === 'function') { onDone(); }
    }

    if (!motionOK) {
      complete();

    }

    gsapLib.to(dom.dock, {
      y: 26,
      opacity: 0,
      duration: 0.4,
      ease: 'power2.in',

      }

  }

  /* ---------------------------------------------------------------------- */
  /* 5. Actions                                                             */
  /* ---------------------------------------------------------------------- */

  function onChangeTheme() {
    hideDock(function () {
      if (dom.cardsWrap) {
        dom.cardsWrap.classList.remove('has-selection');
      }

      updateStatus(null);

      if (motionOK) {
        gsapLib.fromTo(dom.cards, { y: 0 }, {
          y: -7,
          duration: 0.34,
          stagger: 0.05,
          yoyo: true,
          repeat: 1,
          ease: 'power2.inOut',
          onComplete: function () {
            gsapLib.set(dom.cards, { clearProps: 'transform' });
          }
        });
      }

      var focused = null;
      for (var i = 0; i < dom.cards.length; i += 1) {
        if (dom.cards[i].classList.contains('is-selected')) {
          focused = dom.cards[i];
          break;
        }
      }
      if (!focused) { focused = dom.cards[0]; }
      if (focused && typeof focused.focus === 'function') {
        focused.focus({ preventScroll: true });
      }
    });
  }

  function onEnter() {
    if (state.entered) { return; }
    state.entered = true;

    if (!motionOK) {
      window.location.href = ENTRY_URL;
      return;
    }

    var timeline = gsapLib.timeline({
      onComplete: function () {
        window.location.href = ENTRY_URL;
      }
    });

    timeline
      .to(['.page', '.dock', '.footnote'], {
        opacity: 0, y: -16, duration: 0.55, ease: 'power2.inOut'
      }, 0)
      .to('.bg', {
        opacity: 0, duration: 0.7, ease: 'power2.inOut'
      }, 0.08);
  }

  /* ---------------------------------------------------------------------- */
  /* 6. Pointer parallax                                                    */
  /* ---------------------------------------------------------------------- */

  function initParallax() {
    if (!motionOK || !dom.orbs.length) { return; }
    if (window.matchMedia('(pointer: coarse)').matches) { return; }

    var setters = dom.orbs.map(function (orb, index) {
      return {
        x: gsapLib.quickTo(orb, 'x', { duration: 1.4, ease: 'power3.out' }),
        y: gsapLib.quickTo(orb, 'y', { duration: 1.4, ease: 'power3.out' }),
        depth: 16 + index * 12
      };
    });

    var frame = null;
    var pointerX = 0;
    var pointerY = 0;

    window.addEventListener('pointermove', function (event) {
      pointerX = (event.clientX / window.innerWidth - 0.5) * 2;
      pointerY = (event.clientY / window.innerHeight - 0.5) * 2;

      if (frame) { return; }

      frame = window.requestAnimationFrame(function () {
        frame = null;
        for (var i = 0; i < setters.length; i += 1) {
          setters[i].x(pointerX * setters[i].depth);
          setters[i].y(pointerY * setters[i].depth);
        }
      });
    }, { passive: true });
  }

  /* ---------------------------------------------------------------------- */
  /* 7. Events                                                              */
  /* ---------------------------------------------------------------------- */

#!/usr/bin/env bash
# repo-to-txt.sh — dump repo text files into ./EcoFac.txt
set -euo pipefail

if (( BASH_VERSINFO[0] < 4 || (BASH_VERSINFO[0] == 4 && BASH_VERSINFO[1] < 4) )); then
    echo "error: need bash >= 4.4 (found $BASH_VERSION)" >&2; exit 1
fi

REPO_ROOT="${1:-.}"
[[ -d "$REPO_ROOT" ]] || { echo "error: not a directory: $REPO_ROOT" >&2; exit 1; }
cd "$REPO_ROOT"

OUTPUT="portfolio.txt"

# ---- 1. Enumerate -----------------------------------------------------------
# If we're in a git work-tree, git already knows what's "source":
#   --cached            tracked files
#   --others            untracked files…
#   --exclude-standard  …minus anything matched by .gitignore
# .eco.bak, .venv/, __pycache__/, *.pyc, *.egg-info/ are all in .gitignore
# already — git drops them for us. No list to maintain.
if git rev-parse --is-inside-work-tree >/dev/null 2>&1; then
    mapfile -t CANDIDATES < <(git ls-files --cached --others --exclude-standard | sort)
else
    # Fallback for non-git trees — keep the find chain as a safety net.
    mapfile -t CANDIDATES < <(
        find . -type f \
            -not -path './.git/*'                   \
            -not -path '*/.git/*'                   \
            -not -path '*/.eco.bak/*'               \
            -not -path '*/__pycache__/*'            \
            -not -path '*/.mypy_cache/*'            \
            -not -path '*/.pytest_cache/*'          \
            -not -path '*/.ruff_cache/*'            \
            -not -path '*/.tox/*'                   \
            -not -path '*/.eggs/*'                  \
            -not -path '*/.cache/*'                 \
            -not -path '*/.ipynb_checkpoints/*'     \
            -not -path '*/.vscode/*'                \
            -not -path '*/.idea/*'                  \
            -not -path '*/node_modules/*'           \
            -not -path '*/.venv/*'                  \
            -not -path '*/venv/*'                   \
            -not -path '*/env/*'                    \
            -not -path '*/virtualenv/*'             \
            -not -path '*/.virtualenv/*'            \
            -not -path '*/site-packages/*'          \
            -not -path '*/htmlcov/*'                \
            -not -path '*/*.egg-info/*'             \
            -not -path '*/*.dist-info/*'            \
	    -not -path '*/.ecofac.bak'
            -not -path './build/*'                  \
            -not -path './dist/*'                   \
            -not -name '*.pyc'   -not -name '*.pyo' -not -name '*.pyd' \
            -not -name '*.so'    -not -name '*.dylib' -not -name '*.dll' -not -name '*.exe' \
            -not -name '*.egg'   -not -name '.coverage' -not -name 'coverage.xml' \
            -not -name '*.db'    -not -name '*.sqlite' -not -name '*.sqlite3' \
            -not -name '*.log'   -not -name '.DS_Store' \
            -not -name '*.zip'   -not -name '*.tar'   -not -name '*.tgz'  \
            -not -name '*.gz'    -not -name '*.bz2'   -not -name '*.xz'   -not -name '*.7z' \
            -not -name '*.png'   -not -name '*.jpg'   -not -name '*.jpeg' \
            -not -name '*.gif'   -not -name '*.bmp'   -not -name '*.ico'  \
            -not -name '*.webp'  -not -name '*.svg'   -not -name '*.pdf'  \
            -not -name '*.woff'  -not -name '*.woff2' -not -name '*.ttf'  \
            -not -name '*.otf'   -not -name '*.eot'   \
            -not -name '*.mp3'   -not -name '*.mp4'   -not -name '*.mov'  -not -name '*.wav' \
            | sort
    )
fi

# ---- 2. Filter (self, meta, binary sniff) -----------------------------------
FILES=()
for f in "${CANDIDATES[@]}"; do
    # strip leading ./ that find emits, but git ls-files does not
    f="${f#./}"
    [[ -f "$f" ]] || continue
    # never include our own output or the tree dump
    [[ "$f" == "$OUTPUT"  ]] && continue
    [[ "$f" == "tree.txt" ]] && continue
    # binary sniff
    if [[ -s "$f" ]] && ! grep -Iq . -- "$f" 2>/dev/null; then
        continue
    fi
    FILES+=("$f")
done

# ---- 3. Header + index ------------------------------------------------------
{
    echo "# EcoFac — Whole Project Repository Snapshot"
    echo "# Generated : $(date -u +"%Y-%m-%dT%H:%M:%SZ")"
    echo "# Root      : $(pwd)"
    echo "# Files     : ${#FILES[@]}"
    echo ""
    echo "## File Index"
    echo ""
    for f in "${FILES[@]}"; do echo "  $f"; done
    echo ""
} > "$OUTPUT"

# ---- 4. Contents ------------------------------------------------------------
for f in "${FILES[@]}"; do
    {
        echo "======================================================================="
        echo "FILE: $f"
        echo "======================================================================="
        cat -- "$f"
        echo ""
        echo ""
    } >> "$OUTPUT"
done

printf 'Wrote %s — %d files, %s bytes\n' \
       "$OUTPUT" "${#FILES[@]}" "$(wc -c < "$OUTPUT")"

  <script>
  (function () {
    'use strict';

    var reduceQuery = window.matchMedia('(prefers-reduced-motion: reduce)');
    var prefersReduced = reduceQuery.matches;

    if (prefersReduced) {
      document.documentElement.classList.add('no-motion');
    }

    /* ------------------------------------------------------------------ */
    /* Theme switcher (shared system)                                      */
    /* ------------------------------------------------------------------ */

    function initThemeSwitcher() {
      var Theme = window.PortfolioTheme;
      if (!Theme) { return; }

      var buttons = document.querySelectorAll('.nav__theme-btn');

      function syncButtons(active) {
        Array.prototype.forEach.call(buttons, function (btn) {
          var name = btn.getAttribute('data-theme-value');
          btn.setAttribute('aria-pressed', name === active ? 'true' : 'false');
        });
      }

      Array.prototype.forEach.call(buttons, function (btn) {
        btn.addEventListener('click', function () {
          var name = btn.getAttribute('data-theme-value');
          if (Theme.isSupported(name)) { Theme.set(name); }
        });
      });

      syncButtons(Theme.current() || 'dark');

      document.addEventListener('themechange', function (event) {
        syncButtons(event.detail && event.detail.theme
          ? event.detail.theme : 'dark');
      });
    }

    /* ------------------------------------------------------------------ */
    /* Mobile menu                                                         */
    /* ------------------------------------------------------------------ */

    function initMobileMenu() {
      var nav = document.getElementById('nav');
      var toggle = document.getElementById('navToggle');
      var links = document.getElementById('navLinks');
      if (!nav || !toggle || !links) { return; }

      function close() {
        nav.classList.remove('is-open');
        document.body.classList.remove('menu-open');
        toggle.setAttribute('aria-expanded', 'false');
        toggle.setAttribute('aria-label', 'Open menu');
      }
      function open() {
        nav.classList.add('is-open');
        document.body.classList.add('menu-open');
        toggle.setAttribute('aria-expanded', 'true');
        toggle.setAttribute('aria-label', 'Close menu');
      }

      toggle.addEventListener('click', function () {
        if (nav.classList.contains('is-open')) { close(); } else { open(); }
      });

      Array.prototype.forEach.call(links.querySelectorAll('a'), function (a) {
        a.addEventListener('click', close);
      });

      document.addEventListener('keydown', function (event) {
        if (event.key === 'Escape' && nav.classList.contains('is-open')) {
          close();
          toggle.focus();
        }
      });

      window.addEventListener('resize', function () {
        if (window.innerWidth > 780) { close(); }
      });
    }

    /* ------------------------------------------------------------------ */
    /* Nav scroll state + progress bar                                     */
    /* ------------------------------------------------------------------ */

    function initNavScroll() {
      var nav = document.getElementById('nav');
      var progress = document.getElementById('progressBar');
      if (!nav) { return; }

      var ticking = false;

      function update() {
        ticking = false;
        var scrollY = window.scrollY || window.pageYOffset;
        var maxScroll = Math.max(1,
          document.documentElement.scrollHeight - window.innerHeight);

        if (scrollY > 12) { nav.classList.add('is-scrolled'); }
        else { nav.classList.remove('is-scrolled'); }

        if (progress) {
          var pct = Math.min(1, scrollY / maxScroll);
          progress.style.transform = 'scaleX(' + pct.toFixed(4) + ')';
        }
      }

      window.addEventListener('scroll', function () {
        if (!ticking) {
          ticking = true;
          window.requestAnimationFrame(update);
        }
      }, { passive: true });

      update();
    }

    /* ------------------------------------------------------------------ */
    /* Skill cards: expand / collapse                                      */

    function initSkillToggles() {
      var toggles = document.querySelectorAll('.skill__toggle');

      Array.prototype.forEach.call(toggles, function (btn) {
        btn.addEventListener('click', function () {
          var skill = btn.closest('.skill');
          if (!skill) { return; }
          var open = skill.classList.toggle('is-open');
          btn.setAttribute('aria-expanded', open ? 'true' : 'false');
        });
      });
    }

    /* Skill filtering                                                     */
    /* ------------------------------------------------------------------ */

    function initFilters() {
      var filterBtns = document.querySelectorAll('.filter');
      var cards = document.querySelectorAll('#skillsGrid .skill');
      if (!filterBtns.length || !cards.length) { return; }

      function applyFilter(filter, animate) {
        var visible = [];

        Array.prototype.forEach.call(cards, function (card) {
          var match = filter === 'all' ||
                      card.getAttribute('data-category') === filter;
          card.classList.toggle('is-filtered-out', !match);
          if (match) { visible.push(card); }

          /* Collapse expanded panels when filtering */
          card.classList.remove('is-open');
          var t = card.querySelector('.skill__toggle');
          if (t) { t.setAttribute('aria-expanded', 'false'); }
        });

        if (animate && window.gsap && !prefersReduced) {
          gsap.fromTo(visible,
            { opacity: 0, y: 14 },
            {
              opacity: 1,
              y: 0,
              duration: 0.42,
              stagger: 0.035,
              ease: 'power2.out',
              clearProps: 'opacity,transform'
            });
        }
      }

      Array.prototype.forEach.call(filterBtns, function (btn) {
        btn.addEventListener('click', function () {
          var filter = btn.getAttribute('data-filter');

          Array.prototype.forEach.call(filterBtns, function (b) {
            var active = b === btn;
            b.classList.toggle('is-active', active);
            b.setAttribute('aria-pressed', active ? 'true' : 'false');
          });

          applyFilter(filter, true);
        });
      });

      /* Initial state — all visible */
      applyFilter('all', false);
    }

    /* ------------------------------------------------------------------ */
    /* GSAP: scroll reveals + parallax                                     */
    /* ------------------------------------------------------------------ */

    function initAnimations() {
      if (prefersReduced) { return; }

      var gsapLib = window.gsap;
      var ScrollTriggerLib = window.ScrollTrigger;

      if (!gsapLib || !ScrollTriggerLib) {
        document.documentElement.classList.add('no-motion');
        return;
      }

      gsapLib.registerPlugin(ScrollTriggerLib);

      /* Simple per-element reveals */
      var revealEls = document.querySelectorAll('[data-reveal]');
      Array.prototype.forEach.call(revealEls, function (el) {
        gsapLib.to(el, {
          opacity: 1,
          y: 0,
          duration: 0.95,
          ease: 'power3.out',
          scrollTrigger: { trigger: el, start: 'top 88%', once: true }
        });
      });

      /* Skill cards: staggered group reveal */
      var skillCards = document.querySelectorAll('#skillsGrid .skill');
      if (skillCards.length) {
        gsapLib.fromTo(skillCards,
          { opacity: 0, y: 22 },
          {
            opacity: 1,
            y: 0,
            duration: 0.7,
            stagger: 0.035,
            ease: 'power2.out',
            scrollTrigger: {
              trigger: '#skillsGrid',
              start: 'top 82%',
              once: true
            }
          });
      }

      /* Philosophy quote */
      var quote = document.querySelector('.philosophy__quote');
      if (quote) {
        gsapLib.fromTo(quote,
          { opacity: 0, y: 26 },
          {
            opacity: 1,
            y: 0,
            duration: 1.1,
            ease: 'power3.out',
            scrollTrigger: {
              trigger: '.philosophy',
              start: 'top 70%',
              once: true
            }
          });
      }

      /* Orb parallax — desktop non-touch only */
      var orbs = document.querySelectorAll('[data-orb]');
      var isDesktop = window.matchMedia('(min-width: 780px)').matches;
      var isCoarse = window.matchMedia('(pointer: coarse)').matches;

      if (orbs.length && isDesktop && !isCoarse && gsapLib.quickTo) {
        var depths = [18, 24, 12];
        var setters = [];

        for (var i = 0; i < orbs.length; i += 1) {
          setters.push({
            x: gsapLib.quickTo(orbs[i], 'x', { duration: 1.4, ease: 'power3.out' }),
            y: gsapLib.quickTo(orbs[i], 'y', { duration: 1.4, ease: 'power3.out' }),
            depth: depths[i] || 16
          });
        }

        var frame = null;
        var px = 0, py = 0;

        window.addEventListener('pointermove', function (event) {
          px = (event.clientX / window.innerWidth - 0.5) * 2;
          py = (event.clientY / window.innerHeight - 0.5) * 2;

          if (frame) { return; }
          frame = window.requestAnimationFrame(function () {
            frame = null;
            for (var j = 0; j < setters.length; j += 1) {
              setters[j].x(px * setters[j].depth);
              setters[j].y(py * setters[j].depth);
            }
          });
        }, { passive: true });
      }
    }

    /* ------------------------------------------------------------------ */
    /* Boot                                                                */

    function boot() {

      /* Small delay so ScrollTrigger measures positions after layout */
      window.setTimeout(initAnimations, 80);

      /* Watchdog: if GSAP failed to load, unhide revealed content */
      window.setTimeout(function () {
        if (!window.gsap) {
          document.documentElement.classList.add('no-motion');
        }
      }, 2000);
    }

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', boot);
    } else {
      boot();
    }
  })();
  </script>
