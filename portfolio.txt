/* ==========================================================================
   Aman Sohail — Portfolio
   assets/js/site.js
   --------------------------------------------------------------------------
   Shared behaviour for inner pages (about, skills, 404). No dependencies.

     - theme switcher in the nav
     - mobile menu
     - nav background + scroll progress (one rAF-throttled handler)
     - scroll reveals via IntersectionObserver (compositor-driven CSS
       transitions; no GSAP, no leftover inline transforms)
     - orb parallax (CSS variables, eased in a single rAF loop)
   ========================================================================== */

(function (window, document) {
  'use strict';

  var root = document.documentElement;
  var reduced = root.classList.contains('no-motion');

  function $(sel, ctx) { return (ctx || document).querySelector(sel); }
  function $$(sel, ctx) { return Array.prototype.slice.call((ctx || document).querySelectorAll(sel)); }

  /* ---------------------------------------------------------------- Theme */

  function initThemeSwitcher() {
    var Theme = window.PortfolioTheme;
    var buttons = $$('.nav__theme-btn');
    if (!Theme || !buttons.length) { return; }

    function sync(active) {
      buttons.forEach(function (btn) {
        btn.setAttribute('aria-pressed',
          btn.getAttribute('data-theme-value') === active ? 'true' : 'false');
      });
    }

    buttons.forEach(function (btn) {
      btn.addEventListener('click', function () {
        var name = btn.getAttribute('data-theme-value');
        if (Theme.isSupported(name)) { Theme.set(name); }
      });
    });

    sync(Theme.current() || 'dark');

    document.addEventListener('themechange', function (event) {
      sync((event.detail && event.detail.theme) || 'dark');
    });
  }

  /* ---------------------------------------------------------- Mobile menu */

  function initMobileMenu() {
    var nav = $('#nav');
    var toggle = $('#navToggle');
    var links = $('#navLinks');
    if (!nav || !toggle || !links) { return; }

    function setOpen(open) {
      nav.classList.toggle('is-open', open);
      document.body.classList.toggle('menu-open', open);
      toggle.setAttribute('aria-expanded', open ? 'true' : 'false');
      toggle.setAttribute('aria-label', open ? 'Close menu' : 'Open menu');
    }

    toggle.addEventListener('click', function () {
      setOpen(!nav.classList.contains('is-open'));
    });

    $$('a', links).forEach(function (a) {
      a.addEventListener('click', function () { setOpen(false); });
    });

    document.addEventListener('keydown', function (event) {
      if (event.key === 'Escape' && nav.classList.contains('is-open')) {
        setOpen(false);
        toggle.focus();
      }
    });

    window.addEventListener('resize', function () {
      if (window.innerWidth > 780) { setOpen(false); }
    });
  }

  /* ------------------------------------------------- Nav + progress bar */

  function initScrollState() {
    var nav = $('#nav');
    var bar = $('#progressBar');
    if (!nav) { return; }

    var ticking = false;
    var scrolled = false;

    function update() {
      ticking = false;
      var y = window.pageYOffset || root.scrollTop || 0;
      var max = Math.max(1, root.scrollHeight - window.innerHeight);

      var next = y > 12;
      if (next !== scrolled) {
        scrolled = next;
        nav.classList.toggle('is-scrolled', next);
      }
      if (bar) {
        bar.style.transform = 'scaleX(' + Math.min(1, y / max).toFixed(4) + ')';
      }
    }

    window.addEventListener('scroll', function () {
      if (!ticking) {
        ticking = true;
        window.requestAnimationFrame(update);
      }
    }, { passive: true });

    window.addEventListener('resize', update, { passive: true });
    update();
  }

  /* --------------------------------------------------------------- Reveal */

  function finishReveal(el, delay) {
    window.setTimeout(function () {
      el.classList.add('reveal-done');
      el.style.removeProperty('--reveal-delay');
    }, (delay + 0.95) * 1000);
  }

  function initReveal() {
    var items = $$('[data-reveal], [data-reveal-group] > *');
    if (!items.length) { return; }

    if (reduced || !('IntersectionObserver' in window)) {
      items.forEach(function (el) { el.classList.add('is-revealed', 'reveal-done'); });
      return;
    }

    var io = new IntersectionObserver(function (entries) {
      var n = 0;
      entries.forEach(function (entry) {
        if (!entry.isIntersecting) { return; }
        var el = entry.target;
        io.unobserve(el);

        /* Stagger only items that arrive in the same frame. */
        var delay = Math.min(n * 0.07, 0.5);
        n += 1;

        el.style.setProperty('--reveal-delay', delay + 's');
        /* Force a style flush so the delay is in place before the class flips. */
        void el.offsetWidth;
        el.classList.add('is-revealed');
        finishReveal(el, delay);
      });
    }, { rootMargin: '0px 0px -10% 0px', threshold: 0.01 });

    items.forEach(function (el) { io.observe(el); });

    /* Exposed for filters: mark an element as already revealed. */
    window.__revealNow = function (el) {
      io.unobserve(el);
      el.classList.add('is-revealed', 'reveal-done');
    };
  }

  /* ------------------------------------------------------------- Parallax */

  function initParallax() {
    if (reduced) { return; }
    var orbs = $$('[data-orb]');
    if (!orbs.length) { return; }
    if (!window.matchMedia('(min-width: 780px)').matches ||
        window.matchMedia('(pointer: coarse)').matches) { return; }

    var depths = [22, 30, 14];
    var targetX = 0, targetY = 0, curX = 0, curY = 0;
    var running = false;

    function frame() {
      curX += (targetX - curX) * 0.06;
      curY += (targetY - curY) * 0.06;

      orbs.forEach(function (orb, i) {
        var d = depths[i] || 18;
        orb.style.setProperty('--px', (curX * d).toFixed(2) + 'px');
        orb.style.setProperty('--py', (curY * d).toFixed(2) + 'px');
      });

      if (Math.abs(targetX - curX) > 0.001 || Math.abs(targetY - curY) > 0.001) {
        window.requestAnimationFrame(frame);
      } else {
        running = false;
      }
    }

    window.addEventListener('pointermove', function (event) {
      targetX = (event.clientX / window.innerWidth - 0.5) * 2;
      targetY = (event.clientY / window.innerHeight - 0.5) * 2;
      if (!running) {
        running = true;
        window.requestAnimationFrame(frame);
      }
    }, { passive: true });
  }

  /* ----------------------------------------------------------------- Boot */

  function boot() {
    initThemeSwitcher();
    initMobileMenu();
    initScrollState();
    initReveal();
    initParallax();
    root.classList.add('site-ready');
  }

  /* Never leave content hidden if something above throws. */
  function safeBoot() {
    try { boot(); }
    catch (error) {
      root.classList.add('no-motion');
      if (window.console) { window.console.error(error); }
    }
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', safeBoot);
  } else {
    safeBoot();
  }
})(window, document);

(function (window, document) {
  'use strict';

  var reduced = document.documentElement.classList.contains('no-motion');
  var cards = Array.prototype.slice.call(document.querySelectorAll('#skillsGrid .skill'));
  var filters = Array.prototype.slice.call(document.querySelectorAll('.filter'));

  if (!cards.length) { return; }

  /* ---- Expand / collapse ---- */
  cards.forEach(function (card) {
    var btn = card.querySelector('.skill__toggle');
    if (!btn) { return; }
    btn.addEventListener('click', function () {
      var open = card.classList.toggle('is-open');
      btn.setAttribute('aria-expanded', open ? 'true' : 'false');
    });
  });

  /* ---- Counts come from the DOM so they can never drift ---- */
  filters.forEach(function (btn) {
    var key = btn.getAttribute('data-filter');
    var count = btn.querySelector('.filter__count');
    if (!count) { return; }
    count.textContent = key === 'all'
      ? cards.length
      : cards.filter(function (c) { return c.getAttribute('data-category') === key; }).length;
  });

  /* ---- Filtering ---- */
  function applyFilter(key) {
    var shown = 0;

    cards.forEach(function (card) {
      var match = key === 'all' || card.getAttribute('data-category') === key;

      card.classList.remove('is-open', 'is-entering');
      var t = card.querySelector('.skill__toggle');
      if (t) { t.setAttribute('aria-expanded', 'false'); }

      card.classList.toggle('is-filtered-out', !match);
      if (!match) { return; }

      /* Skip the scroll-reveal for cards the visitor is already looking at. */
      if (window.__revealNow) { window.__revealNow(card); }

      if (!reduced) {
        card.style.setProperty('--i', shown);
        /* Restart the animation even if the class was already present. */
        void card.offsetWidth;
        card.classList.add('is-entering');
      }
      shown += 1;
    });
  }

  filters.forEach(function (btn) {
    btn.addEventListener('click', function () {
      filters.forEach(function (b) {
        var active = b === btn;
        b.classList.toggle('is-active', active);
        b.setAttribute('aria-pressed', active ? 'true' : 'false');
      });
      applyFilter(btn.getAttribute('data-filter'));
    });
  });

  /* Clean up the entry animation so it doesn't fight :hover transforms. */
  document.addEventListener('animationend', function (event) {
    if (event.animationName === 'cardIn' && event.target.classList) {
      event.target.classList.remove('is-entering');
    }
  });
})(window, document);

/* ==========================================================================
   Aman Sohail — Portfolio
   assets/js/theme.js
   --------------------------------------------------------------------------
   Shared theme system. Loaded in <head> on every page so the persisted
   theme is applied to <html data-theme="..."> before first paint.

   Also performs the two tiny <html> class updates every page needs before
   first paint, so pages don't need an inline script:
     - no-js  → js          (CSS can hide [data-reveal] only when JS runs)
     - adds    no-motion    when the visitor prefers reduced motion

   Public API (window.PortfolioTheme):
     .KEY                 storage key used for persistence
     .ATTRIBUTE           attribute applied to <html>
     .list                ['dark', 'rich', 'cyberpunk']
     .isSupported(name)   boolean
     .info(name)          { label, color, description } | null
     .label(name)         string
     .get()               persisted theme or null
     .set(name)           persist + apply, returns applied theme
     .apply(name)         apply only (no persistence)
     .clear()             remove preference and reset
     .current()           theme currently applied to <html>
   ========================================================================== */

(function (window, document) {
  'use strict';

  /* Pre-paint <html> flags. */
  (function () {
    var root = document.documentElement;
    root.className = root.className.replace(/\bno-js\b/, 'js');
    if (!/\bjs\b/.test(root.className)) { root.className += ' js'; }
    try {
      if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
        root.className += ' no-motion';
      }
    } catch (error) { /* matchMedia unavailable */ }
  })();

  var STORAGE_KEY = 'amanSohail:theme';
  var ATTRIBUTE = 'data-theme';

  var THEMES = {
    dark: {
      label: 'Dark',
      color: '#0a0a0c',
      description: 'Minimal, focused and professional.'
    },
    rich: {
      label: 'Rich',
      color: '#12072b',
      description: 'Royal purple and polished gold. Regal and authoritative.'
    },
    cyberpunk: {
      label: 'Cyberpunk',
      color: '#07040f',
      description: 'Neon-noir. Acid yellow, hot magenta and electric cyan.'
    }
  };

  var ORDER = ['dark', 'rich', 'cyberpunk'];

  /* ---------------------------------------------------------------------- */

  function isSupported(value) {
    return typeof value === 'string' &&
           Object.prototype.hasOwnProperty.call(THEMES, value);
  }

  function readPreference() {
    try {
      var stored = window.localStorage.getItem(STORAGE_KEY);
      return isSupported(stored) ? stored : null;
    } catch (error) {
      return null; /* private mode / storage disabled */
    }
  }

  function writePreference(value) {
    try {
      window.localStorage.setItem(STORAGE_KEY, value);
      return true;
    } catch (error) {
      return false;
    }
  }

  function removePreference() {
    try {
      window.localStorage.removeItem(STORAGE_KEY);
    } catch (error) {
      /* ignore */
    }
  }

  function updateMetaColor(theme) {
    var meta = document.querySelector('meta[name="theme-color"]');
    if (meta && theme) {
      meta.setAttribute('content', THEMES[theme].color);
    }
  }

  function emit(theme) {
    var event;
    try {
      event = new CustomEvent('themechange', {
        detail: {
          theme: theme,
          label: theme ? THEMES[theme].label : null
        }
      });
    } catch (error) {
      /* CustomEvent constructor unavailable — degrade silently */
      return;
    }
    document.dispatchEvent(event);
  }

  /* ---------------------------------------------------------------------- */

  function apply(value) {
    var theme = isSupported(value) ? value : null;
    var root = document.documentElement;

    if (theme) {
      root.setAttribute(ATTRIBUTE, theme);
      root.classList.add('has-theme');
    } else {
      root.removeAttribute(ATTRIBUTE);
      root.classList.remove('has-theme');
    }

    updateMetaColor(theme);
    emit(theme);

    return theme;
  }

  function set(value) {
    var theme = apply(value);
    if (theme) {
      writePreference(theme);
    } else {
      removePreference();
    }
    return theme;
  }

  function current() {
    var applied = document.documentElement.getAttribute(ATTRIBUTE);
    return isSupported(applied) ? applied : null;
  }

  /* ---------------------------------------------------------------------- */

  /* Apply the stored preference immediately — before the first paint. */
  var initial = readPreference();
  if (initial) {
    apply(initial);
  }

  window.PortfolioTheme = {
    KEY: STORAGE_KEY,
    ATTRIBUTE: ATTRIBUTE,
    list: ORDER.slice(),
    isSupported: isSupported,
    info: function (name) {
      return isSupported(name) ? THEMES[name] : null;
    },
    label: function (name) {
      return isSupported(name) ? THEMES[name].label : '';
    },
    get: readPreference,
    set: set,
    apply: apply,
    clear: function () {
      removePreference();
      return apply(null);
    },
    current: current
  };
})(window, document);

  function cacheDom() {

  function runLoader() {

      if (!loader) { resolve(); return; }

      var settled = false;

      function finish() {
        if (settled) { return; }
        settled = true;

        loader.classList.add('is-done');
        loader.setAttribute('aria-hidden', 'true');
        document.body.classList.remove('is-loading');

        /* Remove from the layout once the fade has completed. */
        window.setTimeout(function () {
          if (loader.parentNode) {
            loader.parentNode.removeChild(loader);
          }
        }, 700);

        resolve();
      }

      /* Failsafe: never hold the visitor hostage. */
      window.setTimeout(finish, 3000);

      /* The full loader plays once per browser session. */
      var seen = false;
      try {
        seen = window.sessionStorage.getItem('amanSohail:loaded') === '1';
        window.sessionStorage.setItem('amanSohail:loaded', '1');
      } catch (error) { /* storage unavailable */ }

      if (!motionOK || seen) {
        if (dom.loaderPct) { dom.loaderPct.textContent = '100'; }
        window.setTimeout(finish, seen ? 140 : 280);
        return;
      }

      var counter = { value: 0 };

      var timeline = gsapLib.timeline({ onComplete: finish });

      timeline
        .from('.loader__mark', {
          y: 14, opacity: 0, duration: 0.7, ease: 'power3.out'
        }, 0.05)
        .from('.loader__track', {
          opacity: 0, scaleX: 0.45, duration: 0.7, ease: 'power3.out'
        }, 0.12)
        .to(dom.loaderFill, {
          scaleX: 1, duration: 0.8, ease: 'power2.inOut'
        }, 0.15)
        .to(counter, {
          value: 100,
          duration: 0.8,
          ease: 'power2.inOut',
          onUpdate: function () {
            if (dom.loaderPct) {
              dom.loaderPct.textContent = pad(Math.round(counter.value));
            }
          }
        }, 0.15)
        .to('.loader__inner', {

  /* ---------------------------------------------------------------------- */

  function selectTheme(name, options) {
    options = options || {};

    var meta = Theme.info(name);
    if (!meta) { return; }

    if (state.selected === name && !options.force) {
      showDock(name, options);
      return;

    state.selected = name;

    dom.cards.forEach(function (card) {
      var active = card.getAttribute('data-theme-value') === name;

    Theme.set(name);

    updateStatus(name, options.returning);

    if (!options.silent) {
      bloom(name);
    }

    showDock(name, { animate: !options.silent, returning: options.returning });

  function bloom(name) {
    if (!motionOK) { return; }

    var card = cardFor(name);
    if (!card) { return; }

    if (dom.flash) {
      var rect = card.getBoundingClientRect();
      var fx = ((rect.left + rect.width / 2) / window.innerWidth) * 100;
      var fy = ((rect.top + rect.height / 2) / window.innerHeight) * 100;

      dom.flash.style.setProperty('--fx', fx.toFixed(2) + '%');
      dom.flash.style.setProperty('--fy', fy.toFixed(2) + '%');

      gsapLib.fromTo(dom.flash, { opacity: 0 }, {
        opacity: 0.26,
        duration: 0.32,
        ease: 'power2.out',
        onComplete: function () {
          gsapLib.to(dom.flash, {
            opacity: 0, duration: 0.9, ease: 'power2.in'
          });
        }
      });
    }

    /* CSS `scale` composes with the card's own transform, so there is no
       snap when the animation ends. */
    card.classList.remove('is-blooming');
    void card.offsetWidth;
    card.classList.add('is-blooming');
    window.setTimeout(function () { card.classList.remove('is-blooming'); }, 700);
  }

    if (!name) {

      dom.status.classList.remove('is-visible');

      gsapLib.fromTo(dom.dock, { y: 34, opacity: 0 }, {
        y: 0,
        opacity: 1,
        duration: 0.85,

  function hideDock(onDone) {
    if (!dom.dock || dom.dock.hidden) {
      if (typeof onDone === 'function') { onDone(); }
      return;
    }

    state.dockVisible = false;
    document.body.classList.remove('dock-open');

    function complete() {
      dom.dock.hidden = true;
      if (typeof onDone === 'function') { onDone(); }

    if (!motionOK) {

  function onChangeTheme() {
    hideDock(function () {
      if (dom.cardsWrap) {
        dom.cardsWrap.classList.remove('has-selection');
      }

      updateStatus(null);

      if (motionOK && dom.cardsWrap) {
        dom.cards.forEach(function (card, i) { card.style.setProperty('--i', i); });
        dom.cardsWrap.classList.remove('is-nudging');
        void dom.cardsWrap.offsetWidth;
        dom.cardsWrap.classList.add('is-nudging');
        window.setTimeout(function () {
          dom.cardsWrap.classList.remove('is-nudging');
        }, 1000);
      }

      var focused = null;
      for (var i = 0; i < dom.cards.length; i += 1) {
        if (dom.cards[i].classList.contains('is-selected')) {
          focused = dom.cards[i];
          break;
        }
      }
      if (!focused) { focused = dom.cards[0]; }
      if (focused && typeof focused.focus === 'function') {
        focused.focus({ preventScroll: true });
      }
    });
  }

  function onEnter() {
    if (state.entered) { return; }
    state.entered = true;

    if (!motionOK) {
      window.location.href = ENTRY_URL;
      return;
    }

    var timeline = gsapLib.timeline({
      onComplete: function () {
        window.location.href = ENTRY_URL;
      }
    });

    timeline
      .to(['.page', '.dock', '.footnote'], {
        opacity: 0, y: -16, duration: 0.55, ease: 'power2.inOut'
      }, 0)
      .to('.bg', {
        opacity: 0, duration: 0.7, ease: 'power2.inOut'
      }, 0.08);
  }

  /* ---------------------------------------------------------------------- */
  /* 6. Pointer parallax                                                    */
  /* ---------------------------------------------------------------------- */

  function initParallax() {
    if (!motionOK || !dom.orbs.length) { return; }
    if (window.matchMedia('(pointer: coarse)').matches) { return; }

    var setters = dom.orbs.map(function (orb, index) {
      return {
        x: gsapLib.quickTo(orb, 'x', { duration: 1.4, ease: 'power3.out' }),
        y: gsapLib.quickTo(orb, 'y', { duration: 1.4, ease: 'power3.out' }),
        depth: 16 + index * 12
      };
    });

    var frame = null;
    var pointerX = 0;
    var pointerY = 0;

    window.addEventListener('pointermove', function (event) {
      pointerX = (event.clientX / window.innerWidth - 0.5) * 2;
      pointerY = (event.clientY / window.innerHeight - 0.5) * 2;

      if (frame) { return; }

      frame = window.requestAnimationFrame(function () {
        frame = null;
        for (var i = 0; i < setters.length; i += 1) {
          setters[i].x(pointerX * setters[i].depth);
          setters[i].y(pointerY * setters[i].depth);
        }
      });
    }, { passive: true });
  }

  /* ---------------------------------------------------------------------- */
  /* 7. Events                                                              */
  /* ---------------------------------------------------------------------- */

  function bindEvents() {
    dom.cards.forEach(function (card) {
      card.addEventListener('click', function () {
        selectTheme(card.getAttribute('data-theme-value'), {});
      });
    });

    if (dom.changeBtn) {
      dom.changeBtn.addEventListener('click', onChangeTheme);
    }

    if (dom.enterBtn) {
      dom.enterBtn.addEventListener('click', onEnter);
    }
  }

  /* ---------------------------------------------------------------------- */
  /* 8. Boot                                                                */
  /* ---------------------------------------------------------------------- */

  function init() {
    cacheDom();

    document.body.classList.add('is-loading');
    document.body.classList.add('is-booting');

    bindEvents();
    initParallax();

    var saved = Theme.get();
    var isReturning = !!saved;

    runLoader().then(reveal).then(function () {
      if (isReturning) {
        /* Restore the previous choice without re-prompting the visitor. */
        selectTheme(saved, {
          silent: true,
          force: true,
          returning: true
        });
      }
    });
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }
})(window, document);

# repo-to-txt.sh — dump repo text files into ./EcoFac.txt
set -euo pipefail

if (( BASH_VERSINFO[0] < 4 || (BASH_VERSINFO[0] == 4 && BASH_VERSINFO[1] < 4) )); then
    echo "error: need bash >= 4.4 (found $BASH_VERSION)" >&2; exit 1
fi

REPO_ROOT="${1:-.}"
[[ -d "$REPO_ROOT" ]] || { echo "error: not a directory: $REPO_ROOT" >&2; exit 1; }
cd "$REPO_ROOT"

OUTPUT="portfolio.txt"

# ---- 1. Enumerate -----------------------------------------------------------
# If we're in a git work-tree, git already knows what's "source":
#   --cached            tracked files
#   --others            untracked files…
#   --exclude-standard  …minus anything matched by .gitignore
# .eco.bak, .venv/, __pycache__/, *.pyc, *.egg-info/ are all in .gitignore
# already — git drops them for us. No list to maintain.
if git rev-parse --is-inside-work-tree >/dev/null 2>&1; then
    mapfile -t CANDIDATES < <(git ls-files --cached --others --exclude-standard | sort)
else
    # Fallback for non-git trees — keep the find chain as a safety net.
    mapfile -t CANDIDATES < <(
        find . -type f \
            -not -path './.git/*'                   \
            -not -path '*/.git/*'                   \
            -not -path '*/.eco.bak/*'               \
            -not -path '*/__pycache__/*'            \
            -not -path '*/.mypy_cache/*'            \
            -not -path '*/.pytest_cache/*'          \
            -not -path '*/.ruff_cache/*'            \
            -not -path '*/.tox/*'                   \
            -not -path '*/.eggs/*'                  \
            -not -path '*/.cache/*'                 \
            -not -path '*/.ipynb_checkpoints/*'     \
            -not -path '*/.vscode/*'                \
            -not -path '*/.idea/*'                  \
            -not -path '*/node_modules/*'           \
            -not -path '*/.venv/*'                  \
            -not -path '*/venv/*'                   \
            -not -path '*/env/*'                    \
            -not -path '*/virtualenv/*'             \
            -not -path '*/.virtualenv/*'            \
            -not -path '*/site-packages/*'          \
            -not -path '*/htmlcov/*'                \
            -not -path '*/*.egg-info/*'             \
            -not -path '*/*.dist-info/*'            \
            -not -path '*/.ecofac.bak/*'            \
            -not -path './build/*'                  \
            -not -path './dist/*'                   \
            -not -name '*.pyc'   -not -name '*.pyo' -not -name '*.pyd' \
            -not -name '*.so'    -not -name '*.dylib' -not -name '*.dll' -not -name '*.exe' \
            -not -name '*.egg'   -not -name '.coverage' -not -name 'coverage.xml' \
            -not -name '*.db'    -not -name '*.sqlite' -not -name '*.sqlite3' \
            -not -name '*.log'   -not -name '.DS_Store' \
            -not -name '*.zip'   -not -name '*.tar'   -not -name '*.tgz'  \
            -not -name '*.gz'    -not -name '*.bz2'   -not -name '*.xz'   -not -name '*.7z' \
            -not -name '*.png'   -not -name '*.jpg'   -not -name '*.jpeg' \
            -not -name '*.gif'   -not -name '*.bmp'   -not -name '*.ico'  \
            -not -name '*.webp'  -not -name '*.svg'   -not -name '*.pdf'  \
            -not -name '*.woff'  -not -name '*.woff2' -not -name '*.ttf'  \
            -not -name '*.otf'   -not -name '*.eot'   \
            -not -name '*.mp3'   -not -name '*.mp4'   -not -name '*.mov'  -not -name '*.wav' \
            | sort
    )
fi

# ---- 2. Filter (self, meta, binary sniff) -----------------------------------
FILES=()
for f in "${CANDIDATES[@]}"; do
    # strip leading ./ that find emits, but git ls-files does not
    f="${f#./}"
    [[ -f "$f" ]] || continue
    # never include our own output or the tree dump
    [[ "$f" == "$OUTPUT"  ]] && continue
    [[ "$f" == "tree.txt" ]] && continue
    # binary sniff
    if [[ -s "$f" ]] && ! grep -Iq . -- "$f" 2>/dev/null; then
        continue
    fi
    FILES+=("$f")
done

# ---- 3. Header + index ------------------------------------------------------
{
    echo "# EcoFac — Whole Project Repository Snapshot"
    echo "# Generated : $(date -u +"%Y-%m-%dT%H:%M:%SZ")"
    echo "# Root      : $(pwd)"
    echo "# Files     : ${#FILES[@]}"
    echo ""
    echo "## File Index"
    echo ""
    for f in "${FILES[@]}"; do echo "  $f"; done
    echo ""
} > "$OUTPUT"

# ---- 4. Contents ------------------------------------------------------------
for f in "${FILES[@]}"; do
    {
        echo "======================================================================="
        echo "FILE: $f"
        echo "======================================================================="
        cat -- "$f"
        echo ""
        echo ""
    } >> "$OUTPUT"
done

printf 'Wrote %s — %d files, %s bytes\n' \
       "$OUTPUT" "${#FILES[@]}" "$(wc -c < "$OUTPUT")"
